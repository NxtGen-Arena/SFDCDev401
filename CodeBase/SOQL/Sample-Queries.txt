1️⃣ Retrieve all Account names and their corresponding Industry
SELECT Name, Industry
FROM Account

2️⃣ Find the total number of Opportunities for each Account
SELECT AccountId, COUNT(Id)
FROM Opportunity
GROUP BY AccountId

3️⃣ List all Closed Won Opportunities along with their Amounts
SELECT Name, Amount
FROM Opportunity
WHERE StageName = 'Closed Won'

4️⃣ Identify the five most recently created Leads
SELECT Name, CreatedDate
FROM Lead
ORDER BY CreatedDate DESC
LIMIT 5

5️⃣ Accounts and Opportunities where Amount > $50,000
SELECT Name,
       (SELECT Name, Amount FROM Opportunities WHERE Amount > 50000)
FROM Account

6️⃣ Find the average Amount of all Closed Won Opportunities
SELECT AVG(Amount)
FROM Opportunity
WHERE StageName = 'Closed Won'

7️⃣ Opportunities closing in the next 7 days
SELECT Name, CloseDate
FROM Opportunity
WHERE CloseDate = NEXT_N_DAYS:7

8️⃣ Contacts related to Opportunities in “Proposal” stage
SELECT Contact.Name, Contact.Email
FROM OpportunityContactRole
WHERE Opportunity.StageName = 'Proposal'

9️⃣ Account with the maximum number of Opportunities
SELECT AccountId, COUNT(Id)
FROM Opportunity
GROUP BY AccountId
ORDER BY COUNT(Id) DESC
LIMIT 1

🔟 Opportunities closing in the current month

FROM Opportunity
